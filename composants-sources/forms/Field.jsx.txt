import React from 'react';
export function Field({label,hint,error,required,htmlFor,children,style}){
  return <div style={{display:'flex',flexDirection:'column',gap:6,...style}}>
    {label&&<label htmlFor={htmlFor} style={{font:'var(--type-label)',color:error?'var(--status-error)':'var(--text-body)',fontWeight:600}}>{label}{required&&<span aria-hidden="true" style={{color:'var(--status-error)'}}> *</span>}</label>}
    {hint&&<span style={{font:'var(--type-body-sm)',color:'var(--text-muted)'}}>{hint}</span>}
    {children}
    {error&&<span role="alert" style={{font:'var(--type-body-sm)',color:'var(--status-error)',display:'flex',gap:6,alignItems:'center',fontWeight:500}}>{error}</span>}
  </div>;
}
