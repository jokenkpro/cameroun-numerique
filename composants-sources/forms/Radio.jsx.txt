import React from 'react';
export function RadioGroup({name,options=[],value,onChange,legend,hint,disabled=false,inline=false,style}){
  return <fieldset style={{border:0,margin:0,padding:0,minWidth:0,...style}}>{legend&&<legend style={{font:'var(--type-label)',fontWeight:600,color:'var(--text-body)',padding:0,marginBottom:hint?2:6}}>{legend}</legend>}{hint&&<div style={{font:'var(--type-body-sm)',color:'var(--text-muted)',marginBottom:6}}>{hint}</div>}<div style={{display:'flex',flexDirection:inline?'row':'column',flexWrap:'wrap',gap:inline?'0 24px':0}}>{options.map(o=>{const opt=typeof o==='string'?{value:o,label:o}:o;return <Radio key={opt.value} name={name} value={opt.value} label={opt.label} hint={opt.hint} checked={value===opt.value} disabled={disabled||opt.disabled} onChange={()=>onChange&&onChange(opt.value)}/>;})}</div></fieldset>;
}
export function Radio({name,value,label,hint,checked=false,disabled=false,onChange}){
  const id=React.useId();
  return <label htmlFor={id} style={{display:'flex',gap:12,alignItems:'flex-start',cursor:disabled?'not-allowed':'pointer',minHeight:'var(--touch-min)',padding:'10px 0'}}>
    <span style={{position:'relative',display:'inline-flex'}}><input id={id} type="radio" name={name} value={value} checked={checked} disabled={disabled} onChange={onChange} style={{position:'absolute',inset:0,opacity:0,margin:0,width:'100%',height:'100%',cursor:'inherit'}}/><span aria-hidden="true" style={{width:24,height:24,borderRadius:'50%',border:'2px solid',borderColor:checked?'var(--cm-vert-800)':'var(--border-input)',background:'var(--surface-card)',display:'inline-flex',alignItems:'center',justifyContent:'center',opacity:disabled?0.5:1,transition:'var(--transition-color)'}}>{checked&&<span style={{width:12,height:12,borderRadius:'50%',background:'var(--cm-vert-800)'}}/>}</span></span>
    <span style={{display:'flex',flexDirection:'column',gap:2}}><span style={{font:'var(--type-body)',lineHeight:'24px',color:disabled?'var(--text-disabled)':'var(--text-body)'}}>{label}</span>{hint&&<span style={{font:'var(--type-body-sm)',color:'var(--text-muted)'}}>{hint}</span>}</span>
  </label>;
}
