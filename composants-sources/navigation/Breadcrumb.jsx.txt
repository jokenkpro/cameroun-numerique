import React from 'react';
import {Icon} from '../core/Icon.jsx';
export function Breadcrumb({items=[],style}){
  return <nav aria-label="Fil d’Ariane" style={style}><ol style={{display:'flex',flexWrap:'wrap',alignItems:'center',gap:6,listStyle:'none',margin:0,padding:0,font:'var(--type-body-sm)'}}>{items.map((it,i)=>{const last=i===items.length-1;return <li key={i} style={{display:'inline-flex',alignItems:'center',gap:6}}>{last?<span aria-current="page" style={{color:'var(--text-muted)'}}>{it.label}</span>:<a href={it.href||'#'} style={{color:'var(--text-link)'}}>{it.label}</a>}{!last&&<Icon name="chevron-right" size={14} style={{color:'var(--text-subtle)'}}/>}</li>;})}</ol></nav>;
}
