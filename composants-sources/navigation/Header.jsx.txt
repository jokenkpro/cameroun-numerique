import React from 'react';
import {Icon} from '../core/Icon.jsx';
import {Button} from '../core/Button.jsx';
import {IconButton} from '../core/IconButton.jsx';
export function Header({serviceName='Portail des services publics',tagline,nav=[],active,onNavigate,lang='FR',onLangChange,user,onSearch,compact=false,showLogin=true,style}){
  const [q,setQ]=React.useState('');const [open,setOpen]=React.useState(false);const [narrow,setNarrow]=React.useState(false);
  React.useEffect(()=>{const mq=window.matchMedia('(max-width: 820px)');const f=()=>setNarrow(mq.matches);f();mq.addEventListener('change',f);return()=>mq.removeEventListener('change',f);},[]);
  React.useEffect(()=>{if(!open)return;const k=e=>{if(e.key==='Escape')setOpen(false);};document.addEventListener('keydown',k);return()=>document.removeEventListener('keydown',k);},[open]);
  return <header style={{background:'var(--surface-card)',borderBottom:'1px solid var(--border-default)',fontFamily:'var(--font-sans)',...style}}>
    <div style={{height:'var(--brand-band-h)',background:'var(--brand-band)'}}/>
    <div style={{maxWidth:'var(--container-lg)',margin:'0 auto',padding:compact?'12px var(--gutter)':'20px var(--gutter)',display:'flex',alignItems:'center',gap:'var(--space-4)',boxSizing:'border-box',flexWrap:'wrap'}}>
      <a href="#" style={{display:'flex',alignItems:'center',gap:16,textDecoration:'none',color:'inherit',flex:'1 1 auto',minWidth:0}}>
        {!narrow&&<span style={{font:'var(--type-overline)',letterSpacing:'var(--tracking-caps)',textTransform:'uppercase',color:'var(--text-heading)',lineHeight:1.3}}>République<br/>du Cameroun</span>}
        {!narrow&&<span style={{width:1,height:36,background:'var(--border-default)'}}/>}
        <span style={{display:'flex',flexDirection:'column',minWidth:0}}><span style={{font:'var(--type-h6)',color:'var(--text-heading)',fontSize:narrow?'var(--text-base)':undefined}}>{serviceName}</span>{tagline&&!compact&&<span style={{font:'var(--type-caption)',color:'var(--text-muted)',fontWeight:400}}>{tagline}</span>}</span>
      </a>
      {!narrow&&<div style={{flex:'1 0 8px'}}/>}
      {onSearch&&!narrow&&<form onSubmit={e=>{e.preventDefault();onSearch(q);}} style={{display:'flex',alignItems:'stretch',flex:'0 1 320px',minWidth:160}}><input value={q} onChange={e=>setQ(e.target.value)} placeholder="Rechercher une démarche" aria-label="Rechercher" style={{flex:1,minWidth:0,height:'var(--control-h-md)',border:'2px solid var(--border-input)',borderRight:0,padding:'0 12px',font:'var(--type-body)',outline:'none',boxSizing:'border-box'}}/><IconButton icon="search" label="Rechercher" variant="primary" type="submit"/></form>}
      {onLangChange&&!narrow&&<button type="button" onClick={()=>onLangChange(lang==='FR'?'EN':'FR')} style={{all:'unset',cursor:'pointer',display:'inline-flex',alignItems:'center',gap:6,font:'var(--type-label)',fontWeight:600,padding:'8px 10px',color:'var(--text-body)',whiteSpace:'nowrap'}}><Icon name="globe" size={16}/>{lang}<span style={{color:'var(--text-subtle)',fontWeight:400}}>· {lang==='FR'?'EN':'FR'}</span></button>}
      {user?<span style={{display:'inline-flex',alignItems:'center',gap:8,font:'var(--type-label)'}}><span style={{width:32,height:32,background:'var(--cm-vert-800)',color:'#fff',display:'inline-flex',alignItems:'center',justifyContent:'center',font:'var(--type-caption)',fontFamily:'var(--font-display)',fontWeight:700}}>{user.initials}</span>{user.name}</span>:showLogin&&!narrow?<Button variant="secondary" size="md" iconLeft="user">Se connecter</Button>:null}
      {narrow&&nav.length>0&&<IconButton icon={open?'x':'menu'} label={open?'Fermer le menu':'Ouvrir le menu'} variant="secondary" aria-expanded={open} aria-controls="header-menu" onClick={()=>setOpen(!open)}/>}
    </div>
    {nav.length>0&&!narrow&&<nav aria-label="Menu principal" style={{borderTop:'1px solid var(--border-default)'}}><ul style={{maxWidth:'var(--container-lg)',margin:'0 auto',padding:'0 var(--gutter)',display:'flex',gap:4,listStyle:'none',boxSizing:'border-box',overflowX:'auto'}}>{nav.map(n=>{const a=n.value===active;return <li key={n.value}><button type="button" onClick={()=>onNavigate&&onNavigate(n.value)} aria-current={a?'page':undefined} style={{all:'unset',cursor:'pointer',display:'block',padding:'14px 12px',font:'var(--type-label)',fontWeight:600,color:a?'var(--cm-vert-800)':'var(--text-body)',borderBottom:'3px solid',borderColor:a?'var(--cm-vert-800)':'transparent',whiteSpace:'nowrap'}}>{n.label}</button></li>;})}</ul></nav>}
    {nav.length>0&&narrow&&open&&<nav id="header-menu" aria-label="Menu principal" style={{borderTop:'1px solid var(--border-default)',background:'var(--surface-card)'}}><ul style={{margin:0,padding:'8px var(--gutter)',listStyle:'none',display:'grid',gap:2}}>{nav.map(n=>{const a=n.value===active;return <li key={n.value}><button type="button" onClick={()=>{setOpen(false);onNavigate&&onNavigate(n.value);}} aria-current={a?'page':undefined} style={{all:'unset',boxSizing:'border-box',cursor:'pointer',display:'block',width:'100%',minHeight:44,padding:'12px 12px',font:'var(--type-label)',fontWeight:600,color:a?'var(--cm-vert-800)':'var(--text-body)',borderLeft:'3px solid',borderColor:a?'var(--cm-vert-800)':'transparent',background:a?'var(--surface-brand-soft)':'transparent'}}>{n.label}</button></li>;})}{onLangChange&&<li><button type="button" onClick={()=>onLangChange(lang==='FR'?'EN':'FR')} style={{all:'unset',boxSizing:'border-box',cursor:'pointer',display:'flex',gap:8,alignItems:'center',width:'100%',minHeight:44,padding:'12px',font:'var(--type-label)',fontWeight:600,borderTop:'1px solid var(--border-default)',marginTop:6}}><Icon name="globe" size={16}/>Langue : {lang} <span style={{color:'var(--text-subtle)',fontWeight:400}}>· passer en {lang==='FR'?'EN':'FR'}</span></button></li>}</ul></nav>}
  </header>;
}
