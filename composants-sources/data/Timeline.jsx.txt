import React from 'react';
import {Icon} from '../core/Icon.jsx';
/** Chronologie d'un dossier ou d'un parcours (J-60 … J-1). */
export function Timeline({items=[],style}){
  const col={done:'var(--cm-vert-800)',current:'var(--cm-jaune-600)',todo:'var(--cm-gris-300)',error:'var(--cm-rouge-700)'};
  return <ol style={{listStyle:'none',margin:0,padding:0,fontFamily:'var(--font-sans)',...style}}>
    {items.map((it,i)=>{const st=it.status||'todo';const last=i===items.length-1;
      return <li key={i} style={{display:'grid',gridTemplateColumns:'auto 1fr',gap:'0 16px'}}>
        <div style={{display:'flex',flexDirection:'column',alignItems:'center'}}>
          <span aria-hidden="true" style={{width:14,height:14,flex:'none',background:st==='todo'?'var(--surface-card)':col[st],border:'2px solid '+col[st],display:'inline-flex',alignItems:'center',justifyContent:'center',marginTop:4}}>{st==='done'&&<Icon name="check" size={9} color="#fff" strokeWidth={3}/>}</span>
          {!last&&<span style={{flex:1,width:2,background:st==='done'?'var(--cm-vert-800)':'var(--cm-gris-200)',minHeight:24}}/>}
        </div>
        <div style={{paddingBottom:last?0:'var(--space-5)',minWidth:0}}>
          <div style={{display:'flex',gap:10,alignItems:'baseline',flexWrap:'wrap'}}>{it.when&&<span style={{font:'var(--type-mono)',fontSize:12,color:'var(--text-muted)'}}>{it.when}</span>}<span style={{font:'var(--type-label)',fontWeight:st==='current'?700:600,color:st==='todo'?'var(--text-muted)':'var(--text-heading)'}}>{it.title}</span></div>
          {it.description&&<div style={{font:'var(--type-body-sm)',color:'var(--text-muted)',marginTop:2}}>{it.description}</div>}
        </div>
      </li>;})}
  </ol>;
}
