import React from 'react';
/** Lucide icon via CDN sprite-less <i> — requires window.lucide (loaded by Icon on demand). */
const LUCIDE='https://unpkg.com/lucide@0.469.0/dist/umd/lucide.min.js';
let loading=null;
function ensureLucide(){if(typeof window==='undefined')return Promise.resolve();if(window.lucide)return Promise.resolve();if(!loading){loading=new Promise(r=>{const s=document.createElement('script');s.src=LUCIDE;s.onload=r;s.onerror=r;document.head.appendChild(s);});}return loading;}
export function Icon({name,size=20,strokeWidth=1.75,color='currentColor',label,style}){
  const ref=React.useRef(null);
  React.useEffect(()=>{let alive=true;ensureLucide().then(()=>{if(!alive||!ref.current||!window.lucide)return;const fn=window.lucide.icons&&window.lucide.icons[toPascal(name)];if(!fn)return;ref.current.innerHTML='';const el=window.lucide.createElement(fn);el.setAttribute('width',size);el.setAttribute('height',size);el.setAttribute('stroke-width',strokeWidth);el.setAttribute('stroke',color);ref.current.appendChild(el);});return()=>{alive=false};},[name,size,strokeWidth,color]);
  return <span ref={ref} role={label?'img':undefined} aria-label={label} aria-hidden={label?undefined:true} style={{display:'inline-flex',width:size,height:size,flex:'none',lineHeight:0,...style}}/>;
}
function toPascal(s){return String(s).split('-').map(p=>p.charAt(0).toUpperCase()+p.slice(1)).join('');}
