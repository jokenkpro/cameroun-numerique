import React from 'react';
const MAP={neutral:['var(--surface-sunken)','var(--text-body)'],success:['var(--status-success-bg)','var(--status-success)'],error:['var(--status-error-bg)','var(--status-error)'],warning:['var(--status-warning-bg)','var(--status-warning)'],info:['var(--status-info-bg)','var(--status-info)'],new:['var(--cm-jaune-200)','var(--cm-jaune-900)'],brand:['var(--cm-vert-800)','var(--cm-blanc)']};
export function Badge({status='neutral',size='md',dot=false,children,style}){
  const [bg,fg]=MAP[status]||MAP.neutral;
  return <span style={{display:'inline-flex',alignItems:'center',gap:6,background:bg,color:fg,fontFamily:'var(--font-display)',fontWeight:700,fontSize:size==='sm'?'var(--text-xs)':'var(--text-sm)',lineHeight:1,letterSpacing:'0.04em',textTransform:'uppercase',padding:size==='sm'?'4px 6px':'6px 8px',whiteSpace:'nowrap',...style}}>{dot&&<span style={{width:6,height:6,borderRadius:'50%',background:'currentColor'}}/>}{children}</span>;
}
