import React from 'react';
import {Icon} from './Icon.jsx';
import {buttonStyles} from './Button.jsx';
const S={sm:'var(--control-h-sm)',md:'var(--control-h-md)',lg:'var(--control-h-lg)'};
export function IconButton({icon,label,variant='tertiary',size='md',disabled=false,style,...rest}){
  const [state,setState]=React.useState('idle');
  const s={...buttonStyles(variant,size,state,disabled),width:S[size],padding:0,...style};
  return <button type="button" aria-label={label} title={label} disabled={disabled} style={s} onMouseEnter={()=>setState('hover')} onMouseLeave={()=>setState('idle')} onMouseDown={()=>setState('active')} onMouseUp={()=>setState('hover')} {...rest}><Icon name={icon} size={size==='sm'?16:size==='lg'?24:20}/></button>;
}
