import React from 'react';
import {Icon} from '../core/Icon.jsx';
/** Barre gouvernementale commune : identique en tête de chaque site de l'État.
 *  - « Ministères & services » : panneau groupé (hub · sites de l'État · services transverses), site courant marqué, lien vers l'annuaire.
 *  - Recherche : typeahead alimenté par `suggest(q)` → [{group,label,hint,href,icon}] ; Entrée ouvre le premier résultat ou `onSearch(q)`. */
export function GovBar({hubLabel='Cameroun numérique',hubHref='#',ministry,sites=[],onSiteChange,onSearch,suggest,searchPlaceholder='Rechercher une démarche, un frais, un site…',user,onLogin,onAccessibility,lang='FR',onLangChange,style}){
  const [open,setOpen]=React.useState(false);const [q,setQ]=React.useState('');const [narrow,setNarrow]=React.useState(false);const [res,setRes]=React.useState([]);const [act,setAct]=React.useState(-1);const [focus,setFocus]=React.useState(false);const [mob,setMob]=React.useState(false);
  const wrap=React.useRef(null);const sref=React.useRef(null);
  React.useEffect(()=>{const mq=window.matchMedia('(max-width: 1000px)');const f=()=>setNarrow(mq.matches);f();mq.addEventListener('change',f);return()=>mq.removeEventListener('change',f);},[]);
  React.useEffect(()=>{const k=e=>{if(e.key==='Escape'){setOpen(false);setFocus(false);setMob(false);}};const c=e=>{if(wrap.current&&!wrap.current.contains(e.target)){setOpen(false);setFocus(false);}};document.addEventListener('keydown',k);document.addEventListener('mousedown',c);return()=>{document.removeEventListener('keydown',k);document.removeEventListener('mousedown',c);};},[]);
  React.useEffect(()=>{if(!suggest||q.trim().length<2){setRes([]);setAct(-1);return;}const t=setTimeout(()=>{setRes(suggest(q).slice(0,9));setAct(-1);},80);return()=>clearTimeout(t);},[q]);
  const btn={all:'unset',boxSizing:'border-box',cursor:'pointer',display:'inline-flex',alignItems:'center',gap:6,height:36,padding:'0 10px',font:'var(--type-caption)',fontWeight:600,color:'var(--cm-papier)',whiteSpace:'nowrap'};
  const go=r=>{if(!r)return;setFocus(false);setMob(false);setQ('');if(r.onSelect)r.onSelect();else if(r.href)window.location.href=r.href;};
  const submit=e=>{e.preventDefault();if(act>=0&&res[act])return go(res[act]);if(res[0]&&res[0].group!=='Voir')return go(res[0]);onSearch&&onSearch(q);setFocus(false);setMob(false);};
  const key=e=>{if(!res.length)return;if(e.key==='ArrowDown'){e.preventDefault();setAct(a=>Math.min(a+1,res.length-1));}else if(e.key==='ArrowUp'){e.preventDefault();setAct(a=>Math.max(a-1,-1));}};
  const groups=[];sites.forEach(s=>{const g=s.group||'Sites de l’État';const x=groups.find(y=>y.g===g);if(x)x.items.push(s);else groups.push({g,items:[s]});});
  const GroupedRes=()=>{const gs=[];res.forEach(r=>{const x=gs.find(y=>y.g===r.group);if(x)x.items.push(r);else gs.push({g:r.group,items:[r]});});let i=-1;return <ul id="govbar-results" role="listbox" aria-label="Suggestions" style={{margin:0,padding:0,listStyle:'none',display:'grid',gap:0}}>{gs.map(gr=><li key={gr.g} role="presentation"><div style={{padding:'10px 12px 4px',font:'var(--type-overline)',letterSpacing:'var(--tracking-caps)',textTransform:'uppercase',color:'var(--text-muted)'}}>{gr.g}</div><ul role="group" style={{margin:0,padding:0,listStyle:'none'}}>{gr.items.map(r=>{i++;const idx=i;return <li key={idx} role="option" aria-selected={act===idx} id={'govbar-opt-'+idx}><button type="button" tabIndex={-1} onMouseEnter={()=>setAct(idx)} onClick={()=>go(r)} style={{all:'unset',boxSizing:'border-box',cursor:'pointer',display:'grid',gridTemplateColumns:'20px 1fr auto',gap:12,alignItems:'center',width:'100%',minHeight:44,padding:'8px 12px',background:act===idx?'var(--surface-brand-soft)':'transparent',borderLeft:'3px solid '+(act===idx?'var(--cm-vert-800)':'transparent')}}><Icon name={r.icon||'arrow-right'} size={16} color="var(--cm-vert-800)"/><span style={{minWidth:0}}><span style={{font:'var(--type-body-sm)',fontWeight:600,color:'var(--text-heading)',display:'block'}}>{r.label}</span>{r.hint&&<span style={{font:'var(--type-caption)',fontWeight:400,color:'var(--text-muted)'}}>{r.hint}</span>}</span>{r.right&&<span style={{font:'var(--type-caption)',color:'var(--text-muted)'}}>{r.right}</span>}</button></li>;})}</ul></li>)}</ul>;};
  const searchBox=(large)=><form role="search" onSubmit={submit} style={{position:'relative',display:'flex',alignItems:'center',height:large?48:30,background:large?'var(--surface-card)':'var(--cm-gris-900)',border:'1px solid '+(focus?'var(--cm-jaune-500)':large?'var(--border-input)':'var(--cm-gris-700)'),flex:large?'1 1 auto':'0 1 300px',minWidth:large?0:160,boxSizing:'border-box'}}><Icon name="search" size={large?18:14} color={large?'var(--text-muted)':'var(--cm-gris-400)'} style={{marginLeft:large?12:8,flex:'none'}}/><input ref={sref} value={q} onChange={e=>setQ(e.target.value)} onFocus={()=>{setFocus(true);setOpen(false);}} onKeyDown={key} role="combobox" aria-expanded={focus&&res.length>0} aria-controls="govbar-results" aria-activedescendant={act>=0?'govbar-opt-'+act:undefined} aria-autocomplete="list" aria-label="Rechercher sur tous les sites de l’État" placeholder={searchPlaceholder} style={{all:'unset',width:'100%',minWidth:0,padding:'0 8px',font:large?'var(--type-body)':'var(--type-caption)',fontWeight:400,color:large?'var(--text-body)':'var(--cm-papier)',boxSizing:'border-box'}}/>{q&&<button type="button" aria-label="Effacer" onClick={()=>{setQ('');sref.current&&sref.current.focus();}} style={{all:'unset',cursor:'pointer',width:32,height:'100%',display:'inline-flex',alignItems:'center',justifyContent:'center',color:large?'var(--text-muted)':'var(--cm-gris-400)'}}><Icon name="x" size={14}/></button>}
    {focus&&(res.length>0||q.trim().length>=2)&&<div style={{position:'absolute',top:'calc(100% + 4px)',left:0,right:0,minWidth:320,background:'var(--surface-card)',color:'var(--text-body)',border:'1px solid var(--border-default)',boxShadow:'var(--shadow-lg)',zIndex:700,maxHeight:'min(70vh,520px)',overflowY:'auto'}}>{res.length>0?<GroupedRes/>:<div style={{padding:'14px 12px',font:'var(--type-body-sm)',color:'var(--text-muted)'}}>Aucune correspondance directe. Entrée : rechercher « {q} » sur tous les sites.</div>}<div style={{borderTop:'1px solid var(--border-default)',padding:'8px 12px',display:'flex',justifyContent:'space-between',gap:8,font:'var(--type-caption)',color:'var(--text-muted)'}}><span>↑↓ pour naviguer · Entrée pour ouvrir</span><button type="button" onClick={()=>{onSearch&&onSearch(q);setFocus(false);setMob(false);}} style={{all:'unset',cursor:'pointer',color:'var(--cm-vert-800)',fontWeight:600}}>Tous les résultats →</button></div></div>}</form>;
  return <div ref={wrap} style={{background:'var(--cm-ebene)',color:'var(--cm-papier)',fontFamily:'var(--font-sans)',position:'relative',zIndex:600,...style}}>
    <div style={{height:'var(--brand-band-h)',background:'var(--brand-band)'}}/>
    <div style={{maxWidth:'var(--container-lg)',margin:'0 auto',padding:'0 var(--gutter)',display:'flex',alignItems:'center',gap:4,minHeight:40,boxSizing:'border-box',flexWrap:'nowrap'}}>
      <a href={hubHref} style={{...btn,paddingLeft:0,textDecoration:'none'}}><span style={{width:3,height:16,background:'var(--brand-band-v)',display:'inline-block'}}/><span style={{fontFamily:'var(--font-display)',letterSpacing:'0.08em',textTransform:'uppercase'}}>{narrow?'Rép. du Cameroun':'République du Cameroun'}</span>{!narrow&&<span className="govbar-hub" style={{color:'var(--cm-gris-400)',fontWeight:400}}>· {hubLabel}</span>}</a>
      {sites.length>0&&<button type="button" aria-expanded={open} aria-controls="govbar-sites" onClick={()=>{setOpen(!open);setFocus(false);}} style={{...btn,background:open?'var(--cm-gris-900)':'transparent',borderBottom:'2px solid '+(open?'var(--cm-jaune-500)':'transparent')}}><Icon name="layout-grid" size={14}/>{narrow?'Services':'Ministères & services'}<Icon name={open?'chevron-up':'chevron-down'} size={14}/></button>}
      <span style={{flex:'1 1 8px',minWidth:0}}/>
      {(onSearch||suggest)&&!narrow&&searchBox(false)}
      {(onSearch||suggest)&&narrow&&<button type="button" aria-label="Rechercher" aria-expanded={mob} onClick={()=>{setMob(!mob);setOpen(false);}} style={btn}><Icon name="search" size={16}/></button>}
      {onAccessibility&&<button type="button" onClick={onAccessibility} aria-label="Accessibilité" style={btn}><Icon name="accessibility" size={16}/>{!narrow&&'Accessibilité'}</button>}
      {onLangChange&&<button type="button" onClick={()=>onLangChange(lang==='FR'?'EN':'FR')} aria-label={'Changer de langue, actuellement '+lang} style={btn}>{lang}<span style={{color:'var(--cm-gris-400)',fontWeight:400}}>· {lang==='FR'?'EN':'FR'}</span></button>}
      {user?<span style={{...btn,cursor:'default'}} title={user.name}><span style={{width:22,height:22,background:'var(--cm-vert-500)',color:'var(--cm-ebene)',display:'inline-flex',alignItems:'center',justifyContent:'center',fontFamily:'var(--font-display)',fontWeight:700,fontSize:11}}>{user.initials}</span>{!narrow&&<span style={{maxWidth:140,overflow:'hidden',textOverflow:'ellipsis',whiteSpace:'nowrap'}}>{user.name}</span>}</span>:<button type="button" onClick={onLogin} style={{...btn,background:'var(--cm-jaune-500)',color:'var(--cm-ebene)',height:28,padding:'0 12px'}}><Icon name="fingerprint" size={14}/>Identité Cameroun</button>}
    </div>
    {mob&&narrow&&<div style={{padding:'8px var(--gutter) 12px',background:'var(--cm-gris-900)'}}>{searchBox(true)}</div>}
    {open&&<div id="govbar-sites" role="region" aria-label="Ministères et services de l’État" style={{position:'absolute',left:0,right:0,top:'100%',background:'var(--surface-card)',color:'var(--text-body)',borderBottom:'1px solid var(--border-default)',boxShadow:'var(--shadow-lg)',zIndex:650}}>
      <div style={{maxWidth:'var(--container-lg)',margin:'0 auto',padding:'var(--space-6) var(--gutter)',boxSizing:'border-box',display:'grid',gridTemplateColumns:'repeat(auto-fit,minmax(240px,1fr))',gap:'var(--space-8)'}}>
        {groups.map(gr=><div key={gr.g}><div style={{font:'var(--type-overline)',letterSpacing:'var(--tracking-caps)',textTransform:'uppercase',color:'var(--accent-laterite)',paddingBottom:8,borderBottom:'1px solid var(--border-default)',marginBottom:6}}>{gr.g}</div><ul style={{margin:0,padding:0,listStyle:'none',display:'grid',gap:2}}>{gr.items.map(s=>{const cur=s.id===ministry;return <li key={s.id}><button type="button" aria-current={cur?'page':undefined} onClick={()=>{setOpen(false);onSiteChange&&onSiteChange(s.id);}} style={{all:'unset',boxSizing:'border-box',cursor:'pointer',display:'grid',gridTemplateColumns:'20px 1fr auto',gap:12,alignItems:'center',width:'100%',minHeight:44,padding:'8px 10px',background:cur?'var(--surface-brand-soft)':'transparent',borderLeft:'3px solid '+(cur?'var(--cm-vert-800)':'transparent')}}><Icon name={s.icon||'globe'} size={16} color="var(--cm-vert-800)"/><span style={{minWidth:0}}><span style={{font:'var(--type-body-sm)',fontWeight:600,color:'var(--text-heading)',display:'block'}}>{s.label}</span>{s.hint&&<span style={{font:'var(--type-caption)',fontWeight:400,color:'var(--text-muted)'}}>{s.hint}</span>}</span>{cur?<span style={{font:'var(--type-caption)',color:'var(--cm-vert-800)',fontWeight:600}}>vous êtes ici</span>:<Icon name="arrow-up-right" size={14} color="var(--text-subtle)"/>}</button></li>;})}</ul></div>)}
      </div>
      <div style={{borderTop:'1px solid var(--border-default)',background:'var(--surface-page)'}}><div style={{maxWidth:'var(--container-lg)',margin:'0 auto',padding:'10px var(--gutter)',boxSizing:'border-box',display:'flex',justifyContent:'space-between',gap:12,flexWrap:'wrap',font:'var(--type-caption)',color:'var(--text-muted)'}}><span><b style={{color:'var(--text-heading)'}}>Un seul réseau.</b> Même barre, même compte Identité Cameroun, mêmes gestes sur chaque site.</span><a href={hubHref} style={{color:'var(--cm-vert-800)',fontWeight:600,textDecoration:'none'}}>Annuaire complet des ministères →</a></div></div>
    </div>}
  </div>;
}
